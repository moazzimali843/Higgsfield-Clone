import { config as loadEnv } from 'dotenv';
import { config, higgsfield } from '@higgsfield/client/v2';

loadEnv({ path: '.env.local', quiet: true });
config({ credentials: process.env.HF_CREDENTIALS });

async function main() {
  const result = await higgsfield.subscribe(
    'bytedance/seedance-2.5/text-to-video',
    {
      input: {
        prompt: 'A cinematic scene at sunset',
        duration: 5,
        resolution: '720p',
        aspect_ratio: '16:9',
        output_format: 'mp4',
        generate_audio: true,
      },
      withPolling: true,
    },
  );

  if (!result.isCompleted) {
    throw new Error('Video generation did not complete.');
  }
  const videoUrl = result.jobs[0]?.results?.raw.url;
  if (!videoUrl) {
    throw new Error('No video URL was returned.');
  }
  console.log(videoUrl);
}

void main();

Here’s what was going on and what we changed.

### How Higgsfield credentials work
The console often shows **one string to copy**, but it is still **two parts**: `api-key-id:api-key-secret`. That matches [Authentication](https://docs.higgsfield.ai/docs/authentication.md) (`Authorization: Key id:secret`) and the SDK env vars `HF_CREDENTIALS` / `HF_KEY` from [open.higgsfield.ai/api-keys](https://open.higgsfield.ai/api-keys). There isn’t a single-token mode in the public docs.

### What the app already did
With **Soul v2** (image) or **Seedance 2.5** (video) selected, the app already called `https://api.higgsfield.ai` via your Next.js routes—not mock data. **Demo** model still uses local demo assets.

### What we updated
1. **One paste field** — Sidebar and image composer use `key-id:key-secret` (same as the Higgsfield integration snippets).
2. **Server env** — `HF_CREDENTIALS` or `HIGGSFIELD_CREDENTIALS` in `.env.local` (split `HIGGSFIELD_KEY_ID` / `SECRET` still work).
3. **Clear auth errors** — Invalid credentials return **401** with messages like `Invalid credentials` instead of silently showing a labeled demo image/video.
4. **API errors** — Reads Higgsfield’s `detail` field (not only `message`).
5. **Video URLs** — Supports both REST `video.url` and SDK-style `jobs[0].results.raw.url`.

### How to get real output
1. Create keys at [open.higgsfield.ai/api-keys](https://open.higgsfield.ai/api-keys).
2. Paste the full `id:secret` string in the sidebar **API key** panel → **Save for session**.
3. On **Image**, pick **Soul v2** (not Demo) and generate.
4. On **Video**, pick **Seedance 2.5** (not Demo) and generate.
